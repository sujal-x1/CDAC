import java.util.List;
import java.util.Optional;

/*
Optional.stream() was introduced in Java 9. 
It converts an Optional into a Stream 
containing zero or one element.

What does Optional.stream() do?

It converts an Optional<T> into a Stream<T>:

The important behavior is:

If Optional contains a value → stream contains 
one element
If Optional is empty → stream contains zero elements
 */

public class Test
{
    public static void main(String[] args)
    {
    	Optional<String> name1 = Optional.of("Nitin");
    	System.out.println(name1.stream().count());   // 1

    	Optional<String> name2 = Optional.empty();
    	System.out.println(name2.stream().count());   // 0
    //Why was it introduced?

// It is particularly useful when working with 
    	// Streams and collections of Optionals.
    	
    	List<Optional<String>> names = List.of(
    		    Optional.of("Nitin"),
    		    Optional.empty(),
    		    Optional.of("Rahul")
    		);

    		List<String> result = names.stream()
    		        .flatMap(Optional::stream)
    		        .toList();

    		System.out.println(result);
    
    }
}
