record Student(int rollno,String name)
{
			
}

/* when you write above code,
the compiler automatically generates this constructor:

public Student(int rollno, String name)
{
    this.rollno = rollno;
    this.name = name;
}

This automatically generated constructor is called the Canonical Constructor.

*/
public class Test 
{
	public static void main(String[] args) 
	{
		Student s1=new Student(1,"Abc");
		var s2=new Student(2,"Xyz");
		System.out.println(s1);
		System.out.println(s2);
System.out.println(s1.name); // private , not accessible
		System.out.println(s1.name());
		System.out.println(s1.rollno());
		System.out.println(s1.equals(s2));
		System.out.println(s1.hashCode());
		System.out.println(s2.hashCode());
		var s3=new Student();  // By default not possible
	}
}
