package mypack;
import java.util.*;
/*
 * accept 5 numbers and display second highest number
 */

public class Test 
{
	public static void main(String[] args) 
	{
		int highest=Integer.MIN_VALUE,secondhighest=Integer.MIN_VALUE, num=0;
		Scanner sc=new Scanner(System.in);
		for(int i=0;i<5;i++)
		{
			num=sc.nextInt();
			if (num > highest)
			{
			    secondhighest = highest;
			    highest = num;
			}
			else if (num > secondhighest)
			{
			    secondhighest = num;
			}
			
		}
		System.out.println("Second Highest number is\t"+secondhighest);
	}
}
