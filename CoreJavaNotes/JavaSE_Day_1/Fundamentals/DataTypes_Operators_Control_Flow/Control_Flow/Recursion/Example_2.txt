package mypack;

import java.util.Scanner;

/*
 * Sum of numbers from 1 to n
 */
public class Test
{
	static int sum(int k)
	{
		if(k==1)
		{
			return 1;
		}
		else
		{
			return k+sum(k-1);
		}
	}

	public static void main(String[] args) 
	{
		System.out.println("Enter a number");
		int n=new Scanner(System.in).nextInt();
		System.out.println(sum(n));
	}
}
