import java.io.IOException;
import java.nio.file.Files;
import java.nio.file.Path;
import java.nio.file.Paths;
import java.util.List;
import java.util.Optional;
import java.util.Scanner;

/*
Files.readString() and Files.writeString() 
were introduced in Java 11. 
They provide simple methods to read 
an entire text file into a String 
and write a String to a file.
*/

public class Test
{
    public static void main(String[] args)
    {
    	// Java 11 introduced:
    	Path path = Path.of("d:\\data.txt");

    	try {
			Files.writeString(path, "Hello Java");
		} catch (IOException e) {
			// TODO Auto-generated catch block
			e.printStackTrace();
		}
    	System.out.println("Now go and check d:\\data.txt");
    	new Scanner(System.in).nextInt();
    	try {
			Files.writeString(path, "Welcome to VITA");
		} catch (IOException e) {
			// TODO Auto-generated catch block
			e.printStackTrace();
		}
    }
}
