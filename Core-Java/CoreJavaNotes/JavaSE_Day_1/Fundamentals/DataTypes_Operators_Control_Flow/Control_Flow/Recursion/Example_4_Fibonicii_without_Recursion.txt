/*
 * Fibonacci - without Recursion
 */
public class Test
{
	public static void main(String[] args) 
	{
		int a=0,b=1,c=0;
		System.out.print(a+"\t");
		System.out.print(b+"\t");
		while(c<30)
		{
			c=a+b;

			a=b;
			b=c;
		}
	}
