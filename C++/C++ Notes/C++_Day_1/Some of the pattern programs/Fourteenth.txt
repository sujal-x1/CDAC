#include<iostream>
using namespace std;

int main()
{
	int i=0,j = 0,sp=5;
	for (i = 1; i <= 5; i++)
	{
		for (sp = 5; sp >= i; sp--)
		{
			cout << " ";
		}
		for (j = 1; j <= i; j++)
		{
			cout << j;
		}
		j -= 2;
		if (i > 1)
		{
			while (j >= 1)
			{
				cout << j--;
			}
		}
		cout << endl;
	}
	for (int i = 4; i>=1; i--)
	{
		for (sp = 5; sp >= i; sp--)
		{
			cout << " ";
		}
		for (j = 1; j <= i; j++)
		{
			cout << j;
		}
		j -= 2;
		if (i > 1)
		{
			while (j >= 1)
			{
				cout << j--;
			}
		}
		cout << endl;
	}
}

/*

	1
       121
      12321
     1234321
    123454321
     1234321
      12321
       121
        1

*/

#include<iostream>
using namespace std;
int main()
{
	int start = 1,end=1,space=5,endspace=5,reverse=false,row=1;
	while (true)
	{
	    
		if (space >= start)
		{
			cout << " ";
			space--;
		}
		else
		{
			if (!reverse)
			{
				if (start <= end)
				{
					cout << start;
					start++;
				}
				else
				{
					start--;
					reverse = true;
				}
			}
			if (start == 1)
			{
				cout << endl;
				row++;
				if (row > 5)
				{
					endspace++;
					end--;
				}
				else
				{
					endspace--;
					end++;
				}
				space = endspace;
				reverse = false;
			}
			if (reverse)
			{
				start--;
				cout << start;
			}
			if (end > 5)
			{
				end = 4;
				start = 1;
				endspace = 1;
			}
		}
		if (row>9)
		{
			break;
		}
	}
}
