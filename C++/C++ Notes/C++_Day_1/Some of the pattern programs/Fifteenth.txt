#include<iostream>
using namespace std;

// if i+j is divisible by 2  display 1 else 0

int main()
{
    for (int i = 1; i < 5; i++)
    {
        for (int j = 1; j <= i; j++)
        {
            if ((i + j) % 2 == 0)
            {
                cout << 1;
            }
            else
            {
                cout << 0;
            }
        }

        cout << endl;
    }
}

#include<iostream>

int main()
{
	int row = 1, count = 0;
	while (true)
	{
		cout << (row+count)%2;
		count++;
		if (count == row)
		{
			cout << endl;
			row++;
			count = 0;
		}
		if (row > 4)
		{
			break;
		}
	}
}
