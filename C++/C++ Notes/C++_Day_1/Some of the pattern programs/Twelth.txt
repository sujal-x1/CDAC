*/

#include<iostream>
using namespace std;
int main()
{
	for (int i = 1; i < 5; i++)
	{
		for (int j = 1; j <= i; j++)
		{
			if (j % 2 == 0)
			{
				cout << j % j;
			}
			else
			{
				cout << j / j;
			}
		}
		cout << endl;
	}
	
}

#include<iostream>
using namespace std;
int main()
{
	int row = 1, count = 0;
	while (true)
	{
		if (count < row)
		{
			count % 2 == 0 ? cout << 1 : cout << 0;
			count++;
		}
		else
		{
			cout << endl;
			row++;
			count = 0;
			if (row > 4)
			{
				break;
			}
		}
	}
	
}
