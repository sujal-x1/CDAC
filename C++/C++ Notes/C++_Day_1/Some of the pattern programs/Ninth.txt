#include<iostream>
using namespace std;
int main()
{
	for (int i=5;i>=1;i--)
	{
		for (int j =5; j>=i; j--)
		{
			cout << j << " ";
		}
		cout << endl;
	}
}


// using single loop


#include<iostream>
using namespace std;
int main()
{
	int start = 5, end = 5;
	while (true)
	{
		if (start>=end)
		{
			cout << start << " ";
			start--;
		}
		else
		{
			cout << endl;
			start = 5;
			end--;
			if (end < 1)
			{
				break;
			}
		}
	}
}
