// using single loop



#include<iostream>
using namespace std;
int main()
{
	char ch = 'A';
	int spacecounter = 0, spacelimit = 7, charlimit = 1, charcounter = 0;
	while (true)
	{
		if (spacecounter<= spacelimit)
		{
			cout << " ";
			spacecounter++;
		}
		else if (ch < 'G')
		{
			cout << ch << " ";
			charcounter++;
			if (charcounter == charlimit)
			{
				cout << endl;
				ch++;
				charlimit++;
				charcounter = 0;
				spacelimit--;
				spacecounter = 0;
			}
		}
		else
		{
			break;
		}
	}
}
