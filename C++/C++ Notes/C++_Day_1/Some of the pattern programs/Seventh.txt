#include<iostream>
using namespace std;
int main()
{
	for (int i=1;i<=5;i++)
	{
		for (int j =5; j >=i; j--)
		{
			cout << j << " ";
		}
		cout << endl;
	}
}

#include<iostream>
using namespace std;
int main()
{
	int start = 5, end = 1;
	while (true)
	{
		if (start >=end)
		{
			cout << start << " ";
			start--;
		}
		else
		{
			cout << endl;
			start = 5;
			end++;
			if (end > start)
			{
				break;
			}
			
		}
	}
}
