#include<iostream>
using namespace std;

// using nested loop


int main()
{
	char ch = '*';
	for (int i = 0; i < 4; i++)
	{
		for (int k = 4; k >= i; k--)
		{
			cout << " ";
		}
		for (int j = 0; j <= i; j++)
		{
			cout << ch << " ";
		}
		cout << endl;
	}
}


// using single loop


int main()
{
	char ch = '*';
	int counter = 0, limit = 1, totalspaces = 8,spacecounter=1;
	for (; ;)
	{
		if (spacecounter < totalspaces)
		{
			cout << " ";
		}
		else
		{
			cout << ch<<"   ";
			counter++;
		}
		spacecounter++;
		if (counter == limit)
		{
			cout << endl;
			totalspaces -= 2;
			limit++;
			counter = 0;
			spacecounter = 1;
		}
		if (totalspaces < 1)
		{
			break;
		}
	}
