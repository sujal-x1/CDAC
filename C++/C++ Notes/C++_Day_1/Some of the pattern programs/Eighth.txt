#include<iostream>
using namespace std;
int main()
{
	for (int i=5;i>=1;i--)
	{
		for (int j =1; j<=i; j++)
		{
			cout << j << " ";
		}
		cout << endl;
	}
}

#include<iostream>
using namespace std;
int main()
{
	int start = 1, end = 5;
	while (true)
	{
		if (start <=end)
		{
			cout << start << " ";
			start++;
		}
		else
		{
			cout << endl;
			start = 1;
			end--;
			if (end < start)
			{
				break;
			}
			
		}
	}
}
