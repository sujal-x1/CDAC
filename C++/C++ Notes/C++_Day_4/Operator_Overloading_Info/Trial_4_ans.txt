// Given the following code, at "here" define a function in such a way which will add m1's num and m2's num and display the result.

#include<iostream>
using namespace std;

class MyClass
{
private:
	int num;
public:
	MyClass(int num)
	{
		this->num = num;
	}
	int getNum()const
	{
		return num;
	}
	friend MyClass add(MyClass&, MyClass&);
};

MyClass add(MyClass& ref1, MyClass& ref2)
{
	return MyClass(ref1.num + ref2.num);
}

int main()
{
	MyClass m1(100), m2(200);
	cout << m1.getNum() << endl;
	cout << m2.getNum() << endl;
	const MyClass &ref=add(m1, m2);
	cout << "ref's data is\t" << ref.getNum() << endl;
}
