// Given the following code, at "here" 
// define a function in such a way which will add 
// m1's num and m2's num and display the result.

#include<iostream>
using namespace std;

class MyClass
{
private:
	int num;
public:
	MyClass(int num)
	{
		this->num = num;
	}
	int getNum()const
	{
		return num;
	}
	
	~MyClass()
	{
		cout << "in MyClass destructor" << endl;
	}
	MyClass& operator++()
	{
		num++;
		return *this;
	}
	MyClass operator++(int k)
	{
		return MyClass(num++);
	}
};

int main()
{
	MyClass m1(100), m2(200);
	cout << m1.getNum() << endl;
	cout << m2.getNum() << endl;
	m2=++m1;  // m1.operator++()
	cout << "m1's data now is\t" << m1.getNum() << endl;
	cout << "m2's data now is\t" << m2.getNum() << endl;

	m1=m2++; // m2.operator++(0)
	cout << "m2's data now is\t" << m2.getNum() << endl;
	cout << "m1's data now is\t" << m1.getNum() << endl;
}
