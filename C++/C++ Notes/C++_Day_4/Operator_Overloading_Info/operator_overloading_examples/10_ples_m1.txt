// Given the following code, at "here" 
// define a function in such a way which will add 
// m1's num and m2's num and display the result.

#include<iostream>
using namespace std;

class MyClass
{
private:
	int num;
public:
	MyClass(int num)
	{
		this->num = num;
	}
	int getNum()const
	{
		return num;
	}
	friend MyClass operator+(int, MyClass&);
};
MyClass operator+(int k, MyClass& ref)
{
	return MyClass(k + ref.num);
}


int main()
{
	MyClass m1(100), m2(200);
	cout << m1.getNum() << endl;
	cout << m2.getNum() << endl;
	const MyClass &ref=10 + m1; // operator+(10,m1)
	cout << "ref's data is\t" << ref.getNum() <<endl;
	
}
