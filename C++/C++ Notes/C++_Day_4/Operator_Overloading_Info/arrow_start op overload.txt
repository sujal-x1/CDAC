/*


m1->getNum()
	what do we need before ->
		pointer of m1 (calling object)
			return this

(*m1).getNum() 
	What do we need before .
		reference of m1 (calling object)
			return *this


*/


#include<iostream>
using namespace std;
class myclass
{
public:
void disp()
{
	cout<<"myclass disp"<<endl;
}
myclass* operator->()
{
	return this;
}
myclass& operator*()
{
	return *this;
}
};
int main()
{
	myclass m;
	m.disp();
	m->disp();
	(*m).disp();
	return 0;
}
