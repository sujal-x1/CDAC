Q1. Registration form validation (10)
Create a form with User ID, Name, Email, Password and Confirm Password. Use onsubmit to validate everything and onchange on individual fields.

No field may be empty (2)
User ID: 5 to 15 characters, only A-Z, a-z, 0-9 (no underscore or special symbols) (3)
Password: 8 to 15 characters (2)
Password and Confirm Password must match (2)
Show error messages next to fields; block submission if invalid (1) 

Soln : 
<body>
    <form onsubmit="return check();">
      Userid :
      <input type="text" name="userid" id="uid" onchange="useridCheck()" /><span
        id="uiderr"
        class="err"
      ></span>

      Email :
      <input
        type="email"
        name="email"
        id="email"
        required
        style="margin-left: 10px"
      />
      <br />

      Name :
      <input
        type="text"
        name="name"
        id="name"
        required
        style="margin-left: 10px"
      />
      <br />

      Password :
      <input type="text" name="pass" id="pass" onchange="passCheck()" />

      Confirm Password :
      <input
        type="text"
        name="confirmpass"
        id="confirmpass"
        onchange="passCheck()"
      />
      <span id="passerr" class="err"></span>

      <input type="submit" value="Submit" />
    </form>
    <script>
      function check() {
        if (useridCheck() && passCheck()) return true;
        else return false;
      }

      function useridCheck() {
        const userid = document.getElementById("uid");
        console.log("reuslt ", /^[A-Za-z0-9]+$/.test(userid.value));
        if (userid.value.length == 0) {
          document.getElementById("uiderr").innerHTML =
            "User ID should not be empty";
          return false;
        } else if (userid.value.length < 5 || userid.value.length > 15) {
          document.getElementById("uiderr").innerHTML = "Min : 8 Max :15";
          return false;
        } else if (!/^[A-Za-z0-9]+$/.test(userid.value)) {
          document.getElementById("uiderr").innerHTML =
            "No Special Chracter allowed";
          return false;
        } else {
          document.getElementById("uiderr").innerHTML = " ";
          return true;
        }
      }

      function passCheck() {
        const pass = document.getElementById("pass");
        const confirmPass = document.getElementById("confirmpass");
        const span = document.getElementById("passerr");
        if (pass.value.length < 8 || pass.value.length > 15) {
          span.innerHTML = "Min : 8 Max : 15";
          return false;
        } else if (pass.value != confirmPass.value) {
          span.innerHTML = "Password does not match";
          return false;
        } else {
          span.innerHTML = " ";
          return true;
        }
      }

    </script>
  </body>

      async function getOne() {
        let id = document.getElementById("getone").value;

        let user = data.find((item) => item.id == id);

          document.getElementById("details").innerHTML = "";
